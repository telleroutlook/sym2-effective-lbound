\begin{abstract}
We prove Theorem~F-2: the global Jacquet--Shalika quadratic integral possesses a
simple pole at $s=1$ with a strictly positive residue.  The positivity is a product
of three factors, each strictly positive:
(1) the Fourier transform $\hat\Phi(0) > 0$,
(2) the residue $\Res_{s=1} \zeta_F(s) > 0$ (Analytic Class Number Formula),
(3) $L(1, \pi, \Ad) > 0$ (Shahidi 1981).
This theorem plays a structural role in the Siegel zero exclusion argument
(proof strategy Case 2 in spec/SPECIFICATION.md).
\end{abstract}

\section{The Jacquet--Shalika Transfer Operator}

Let $\pi = \bigotimes_v' \pi_v$ be an irreducible generic cuspidal automorphic
representation of $GL_2(\A_F)$, with contragredient $\tilde\pi$.  Fix a non-trivial
additive character $\psi: F \backslash \A_F \to \mathbb{C}^\times$.

\begin{definition}[Jacquet--Shalika integral]
For $W_f \in \calW(\pi, \psi)$, $s \in \mathbb{C}$ with $\mathrm{Re}(s) > 1$, and
$\Phi \in \mathcal{S}(\A_F^2)$, define
\[
  \mathcal{T}_{s,\Phi}^{\mathrm{JS}}[W_f \otimes W_{\bar f}]
  = \int_{N_2(\A_F) \backslash GL_2(\A_F)}
    W_f(g)\, \overline{W_f(g)}\, \Phi(e_2 g)\, |\det g|^s_{\A_F}\, d^\times g,
\]
where $e_2 = (0,1) \in F^2$.
\end{definition}

\section{Euler Product Factorization}

\begin{theorem}[Jacquet--Shalika \cite{JS81}]\label{thm:euler-product}
For $\Phi = \prod_v \Phi_v$ a pure tensor and $W_f = \prod_v W_{f,v}$,
\[
  \langle 1, \mathcal{T}_{s,\Phi}^{\mathrm{JS}}[W_f \otimes W_{\bar f}]\rangle
  = \hat\Phi(0) \cdot \zeta_F(s) \cdot L(s, \pi, \Ad) \cdot \|W_f\|_{\calW}^2
  \cdot \prod_{v \in S} \epsilon_v(s),
\]
where $S$ is the finite set of places where $\pi_v$ or $\psi_v$ is ramified,
$\hat\Phi$ is the Fourier transform of $\Phi$, and
$\epsilon_v(s)$ are explicit non-zero local correction factors for $v \in S$.
\end{theorem}

\begin{remark}
For the standard choice $\Phi_v = \mathbf{1}_{\calO_v^2}$ at unramified places,
the correction factors $\epsilon_v(s) = 1$ by Theorem~F-1 (proof/01-foundations.tex).
At Archimedean places, choosing $\Phi_v(x) = e^{-\pi \|x\|_v^2}$ (Gaussian) gives
$\hat\Phi_v(0) = 1$ and $\epsilon_v(1) > 0$ by standard Mellin--Whittaker analysis.
\end{remark}

\section{Positivity of Each Factor}

\subsection{Fourier Transform Term}

\begin{lemma}\label{lem:fourier-pos}
For the Schwartz--Bruhat test function $\Phi = \prod_v \Phi_v$ with $\Phi_v$ as
in the Remark above, $\hat\Phi(0) = \prod_v \hat\Phi_v(0) > 0$.
\end{lemma}

\begin{proof}
At non-Archimedean places, $\hat\Phi_v(0) = \vol(\calO_v^2) = 1$ (normalized Haar
measure).  At Archimedean places, $\hat\Phi_v(0) = \int e^{-\pi\|x\|^2} dx = 1$.
The Euler product converges and equals 1.
\end{proof}

\subsection{Dedekind Zeta Residue}

\begin{theorem}[Analytic Class Number Formula]\label{thm:acnf}
$\zeta_F(s)$ has a simple pole at $s=1$ with strictly positive residue:
\[
  \Res_{s=1} \zeta_F(s)
  = \frac{2^{r_1}(2\pi)^{r_2} h_F R_F}{w_F \sqrt{|D_F|}} > 0,
\]
where $r_1, r_2$ are the numbers of real and complex places, $h_F$ is the class number,
$R_F$ is the regulator, $w_F$ is the number of roots of unity, and $D_F$ is the discriminant.
\end{theorem}

\begin{proof}
Standard; see \cite[Ch.~7]{Neukirch}.  All quantities $h_F, R_F, w_F, |D_F|$ are
positive integers, and $2^{r_1}(2\pi)^{r_2} > 0$.
\end{proof}

\subsection{Non-Vanishing of the Adjoint $L$-Function}

\begin{theorem}[Shahidi \cite{Sh81}]\label{thm:shahidi}
For any irreducible generic cuspidal automorphic representation $\pi$ of $GL_2(\A_F)$,
\[
  L(1, \pi, \Ad) > 0.
\]
\end{theorem}

\begin{proof}
Shahidi (1981) proves that $L(s, \pi, \Ad)$ is entire and non-zero on $\mathrm{Re}(s) = 1$
via the Langlands--Shahidi method: the adjoint $L$-function appears as a normalizing
factor of an Eisenstein series on $GL_3$, and the non-vanishing follows from
the theory of intertwining operators.
\end{proof}

\subsection{Whittaker Norm}

\begin{lemma}\label{lem:norm-pos}
For any non-zero $W_f \in \calW(\pi, \psi)$, $\|W_f\|_{\calW}^2 > 0$.
\end{lemma}

\begin{proof}
The Whittaker $L^2$-norm $\|W_f\|^2 = \int_{N_2(\A_F) \backslash GL_2(\A_F)^1}
|W_f(g)|^2 dg$ is a positive-definite Hermitian form by unitarity of $\pi$.
\end{proof}

\section{Main Theorem: Global Residue Positivity}

\begin{theorem}[F-2: Global Residue Positivity]\label{thm:F2}
For any non-zero $W_f \in \calW(\pi, \psi)$ and $\Phi$ as above,
\[
  \Res_{s=1}\langle 1,\mathcal{T}_{s,\Phi}^{\mathrm{JS}}[W_f \otimes W_{\bar f}]\rangle
  = \hat\Phi(0) \cdot \Res_{s=1}\zeta_F(s) \cdot L(1,\pi,\Ad) \cdot \|W_f\|_{\calW}^2 > 0.
\]
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:euler-product}, the global integral equals
$\hat\Phi(0) \cdot \zeta_F(s) \cdot L(s,\pi,\Ad) \cdot \|W_f\|^2 \cdot \prod_{v \in S}\epsilon_v(s)$
for $\mathrm{Re}(s)$ sufficiently large.  Both sides have meromorphic continuation.
The only simple pole comes from $\zeta_F(s)$ at $s=1$, since $L(s,\pi,\Ad)$ is
holomorphic at $s=1$ by \cite{Sh81} and $\epsilon_v(1) \neq 0$ at ramified places.
Taking the residue:
\begin{align*}
  \Res_{s=1}\langle 1, \mathcal{T}^{\mathrm{JS}}\rangle
  &= \hat\Phi(0) \cdot \bigl(\Res_{s=1}\zeta_F(s)\bigr) \cdot L(1,\pi,\Ad)
     \cdot \|W_f\|^2 \cdot \prod_{v\in S}\epsilon_v(1).
\end{align*}
Each factor is strictly positive:
\begin{itemize}
  \item $\hat\Phi(0) > 0$: Lemma~\ref{lem:fourier-pos}.
  \item $\Res_{s=1}\zeta_F(s) > 0$: Theorem~\ref{thm:acnf}.
  \item $L(1,\pi,\Ad) > 0$: Theorem~\ref{thm:shahidi}.
  \item $\|W_f\|^2 > 0$: Lemma~\ref{lem:norm-pos}.
  \item $\epsilon_v(1) > 0$ for $v \in S$: local non-vanishing at $s=1$,
        proved case-by-case from the JPSS theory of essential vectors.
\end{itemize}
The product is therefore strictly positive.
\end{proof}

\begin{corollary}[Application to Siegel zero exclusion]
If there existed a real zero $\beta \in (0,1)$ of $L(s, \pi, \Ad)$ with $\beta \to 1$
as $N(\pi) \to \infty$, then $L(1, \pi, \Ad) \to 0$, contradicting Theorem~\ref{thm:F2}
(since the residue would also approach zero while the other factors remain bounded
away from zero).  This is the mechanism by which Theorem~F-2 excludes Siegel zeros
in the Goldfeld--Hoffstein--Lieman argument.
\end{corollary}
