\begin{abstract}
We prove the global residue positivity theorem (F-2): the Jacquet--Shalika quadratic
integral has a simple pole at $s=1$ with strictly positive residue. The proof assembles
the local Euler factor factorization (F-1), the Analytic Class Number Formula, and
Shahidi's non-vanishing theorem for the adjoint $L$-function.
\end{abstract}

\section{Global Setup}

\subsection{The Jacquet--Shalika Global Integral}

Let $\pi = \bigotimes_v' \pi_v$ be an irreducible generic cuspidal automorphic
representation of $GL_2(\mathbb{A}_F)$.
For $W_f\in\mathcal{W}(\pi,\psi)$, $\Phi\in\mathcal{S}(\mathbb{A}_F^2)$, define
\begin{equation}\label{eq:global-integral}
  Z(s, W_f, \Phi)
  = \int_{N(\mathbb{A}_F)\backslash GL_2(\mathbb{A}_F)}
    W_f(g)\,\overline{W_f(g)}\,\Phi(e_2\,g)\,|\det g|^s\,d^\times g.
\end{equation}

\subsection{Euler Product Factorization}

\begin{theorem}[Jacquet--Shalika 1981, \texttt{[BASE]}]\label{thm:JS}
For $\mathrm{Re}(s)$ sufficiently large and $\Phi = \prod_v \Phi_v$ factored:
\[
  Z(s, W_f, \Phi) = \hat\Phi(0)\cdot \|W_f\|_\mathcal{W}^2
  \cdot \prod_v I_v(s, W_{f,v}, \Phi_v).
\]
\end{theorem}

\subsection{Global Euler Product via F-1}

At unramified places $v\notin S$, Theorem~F-1 gives
$I_v(s) = \zeta_v(s)\cdot L_v(s,\pi_v,\mathrm{Ad})$.
At finitely many places $v\in S$, local analysis shows $I_v(1) > 0$.
Assembling:
\begin{equation}\label{eq:global-product}
  Z(s, W_f, \Phi)
  = \hat\Phi(0)\cdot\|W_f\|_\mathcal{W}^2
    \cdot \zeta_F(s)\cdot L(s,\pi,\mathrm{Ad})\cdot H(s),
\end{equation}
where $H(s)$ is holomorphic and non-zero at $s=1$.

\section{The Simple Pole and Its Residue}

\subsection{Analytic Class Number Formula}

\begin{theorem}[ACNF, \texttt{[BASE]}]\label{thm:ACNF}
$\mathrm{Res}_{s=1}\zeta_F(s) = \kappa_F = \dfrac{2^{r_1}(2\pi)^{r_2} h_F R_F}{w_F\sqrt{|D_F|}} > 0$.
\end{theorem}

\subsection{Shahidi Non-Vanishing}

\begin{theorem}[Shahidi 1981, \texttt{[BASE]}]\label{thm:Shahidi}
For any irreducible generic cuspidal automorphic $\pi$: $L(1, \pi, \mathrm{Ad}) > 0$.
\end{theorem}

\subsection{Residue Computation}

\begin{theorem}[F-2: Global Residue Positivity \texttt{[THM]}]\label{thm:F2}
For any non-zero $W_f$ and $\Phi$ with $\hat\Phi(0) > 0$:
\[
  \mathrm{Res}_{s=1}\, Z(s, W_f, \Phi)
  = \hat\Phi(0)\cdot\kappa_F\cdot L(1,\pi,\mathrm{Ad})\cdot\|W_f\|_\mathcal{W}^2 > 0.
\]
\end{theorem}
\begin{proof}
From \eqref{eq:global-product}, $Z = \hat\Phi(0)\cdot\|W_f\|^2\cdot\zeta_F(s)\cdot L(s,\pi,\mathrm{Ad})\cdot H(s)$.
Since $L(s,\pi,\mathrm{Ad})$ is holomorphic at $s=1$ (Shimura 1975)
and $L(1,\pi,\mathrm{Ad}) > 0$ by Theorem~\ref{thm:Shahidi},
and $H(1) > 0$:
\[
  \mathrm{Res}_{s=1}\,Z = \hat\Phi(0)\cdot\|W_f\|^2
    \cdot \underbrace{\kappa_F}_{>0}\cdot \underbrace{L(1,\pi,\mathrm{Ad})}_{>0}\cdot \underbrace{H(1)}_{>0} > 0. \quad\square
\]
\end{proof}

\begin{corollary}\label{cor:sym2-positive}
For any primitive Hecke eigenform $f$ with trivial central character:
$L(1, \mathrm{sym}^2 f) > 0$.
\end{corollary}

\begin{remark}[Role in the Siegel zero argument]
Corollary~\ref{cor:sym2-positive} eliminates Case~2 in the
Goldfeld--Hoffstein--Lieman two-case argument: a Siegel zero $\beta\approx 1$
would force $L(1,\mathrm{sym}^2 f)\to 0$, contradicting $L(1,\mathrm{sym}^2 f) > 0$.
This is why the effective lower bound (E-1) reduces to Case~1 only.
\end{remark}

\section{Numerical Instance}

For $f = \Delta\in S_{12}(SL_2(\mathbb{Z}))$ and $F=\mathbb{Q}$:
$\kappa_\mathbb{Q} = 1$, $L(1,\mathrm{sym}^2\Delta) \approx 2.40578$ (see F-3),
and $\mathrm{Res}_{s=1}\,Z = 2.40578\cdot\|W_\Delta\|^2 > 0$.
