\begin{abstract}
This document states the main research goal and its proof strategy:
for any normalized primitive Hecke eigenform $f$ of prime level $p$ with trivial
nebentypus, prove
\[
  L(1, \mathrm{sym}^2 f) \geq \frac{c_{\mathrm{eff}}}{\log p}
\]
with an explicit, computable constant $c_{\mathrm{eff}} > 0$.
The proof follows Goldfeld--Hoffstein--Lieman (1994), but with all implicit constants
made explicit via rigorous interval arithmetic.  The key novelty is a computer-certified
value of $c_{\mathrm{eff}}$.
\end{abstract}

\section{Main Theorem Statement [OBL]}

\begin{theorem}[Main, [OBL]]\label{thm:main}
Let $f \in S_k(p)$ be a normalized primitive Hecke eigenform of weight $k \geq 2$
and prime level $p$.  Then
\[
  L(1, \mathrm{sym}^2 f) \geq \frac{c_{\mathrm{eff}}}{\log p},
\]
where $c_{\mathrm{eff}}$ is an explicit positive constant, independent of $f$ and $p$,
certified by interval arithmetic in \texttt{src/numerical\_delta.py}.
\end{theorem}

\section{Proof Strategy: Goldfeld--Hoffstein--Lieman Dichotomy}

\begin{definition}[Siegel zero]
A Siegel zero for $L(s, \mathrm{sym}^2 f)$ is a real zero $\beta$ satisfying
\[
  \beta > 1 - \frac{c_0}{\log(p(2+|t|))}
\]
for a constant $c_0 > 0$ to be made explicit.
\end{definition}

\subsection{Case 1: No Siegel Zero}

\begin{proposition}[Case 1: explicit bound from zero-free region, [OBL]]\label{prop:case1}
Assume $L(s, \mathrm{sym}^2 f)$ has no Siegel zero.  Then for an explicit $c_1 > 0$:
\[
  L(1, \mathrm{sym}^2 f) \geq \frac{c_1}{\log p}.
\]
\end{proposition}

\begin{proof}[Proof strategy]
From the zero-free region of width $\delta = c_0/\log p$, the Hadamard product
representation of $L(s, \mathrm{sym}^2 f)$ near $s = 1$ gives
\[
  -\frac{L'}{L}(s, \mathrm{sym}^2 f) = \sum_\rho \frac{1}{s-\rho} + O(\log p)
\]
with the sum over zeros controlled by the zero-free region.  Integrating along the
segment $[1-\delta, 1]$ and exponentiating yields
\[
  \log L(1, \mathrm{sym}^2 f) \geq -C \log\log p
\]
for an explicit $C$, giving $L(1, \cdot) \geq (\log p)^{-C}$.  The stronger bound
$c_1/\log p$ (with explicit $c_1$) follows from the full Goldfeld--Hoffstein--Lieman
calculation, which requires tracing the constant in the log-free zero-free region.
\textbf{[OBL: close this argument with explicit $c_1$.]}
\end{proof}

\subsection{Case 2: Siegel Zero Exists -- Excluded by Theorem F-2}

\begin{proposition}[Case 2: Siegel zero excluded]\label{prop:case2}
If $\beta$ is a Siegel zero for $L(s, \mathrm{sym}^2 f)$ with $\beta > 1 - c_0/\log p$,
then $L(1, \mathrm{sym}^2 f) \geq c_2/\log p$ for an explicit $c_2 > 0$.
\end{proposition}

\begin{proof}[Proof strategy]
The existence of a Siegel zero $\beta$ close to 1 forces $L(\beta, \mathrm{sym}^2 f) = 0$,
but by Theorem~F-2 (global residue positivity), $L(1, \pi, \Ad) > 0$ (which equals
$L(1, \mathrm{sym}^2 f)$ for trivial central character).  More precisely, for
$\beta = 1 - \varepsilon$ with $\varepsilon = O(1/\log p)$:
\[
  L(1, \mathrm{sym}^2 f) = L(\beta, \cdot) + L'(\beta', \cdot)(1-\beta)
  \geq -L'(\beta', \cdot) \cdot \varepsilon
\]
for some $\beta' \in (\beta, 1)$, where $-L'(\beta', \cdot)$ can be bounded below
using the Rankin--Selberg method from Theorem~F-2.
This gives $L(1, \cdot) \geq c_2 \varepsilon = c_2/\log p$.
\textbf{[OBL: make $c_2$ explicit from Theorem~F-2 quantitative bounds.]}\end{proof}

\subsection{Conclusion}

Combining Cases 1 and 2 with $c_{\mathrm{eff}} = \min(c_1, c_2)$ yields
Theorem~\ref{thm:main}.  The constant $c_{\mathrm{eff}}$ will be certified by
interval arithmetic once the two case constants are computed.

\section{Explicit Constant Computation Plan [OBL]}

To compute $c_{\mathrm{eff}}$ explicitly, we need:

\begin{enumerate}
  \item \textbf{Analytic conductor bound.}  For $f \in S_k(p)$, the analytic conductor of
        $\Pi = \mathrm{sym}^2 f$ satisfies $q(\Pi) \leq Cp^2$ for an explicit $C$
        (depending only on the weight $k$).  This follows from the conductor formula
        for symmetric square lifts via the Gelbart--Jacquet theory.
  \item \textbf{Log-free zero-free region.}  An explicit version of the Vinogradov--Korobov
        zero-free region for $GL_3$ $L$-functions gives: $L(s, \Pi) \neq 0$ for
        $\mathrm{Re}(s) > 1 - c_3 / (\log q(\Pi) \cdot \log\log q(\Pi))^{2/3}$
        with explicit $c_3$ from \cite{Vin58,Kor58}.
  \item \textbf{Hadamard product constants.}  The functional equation for $L(s, \Pi)$
        (Shimura \cite{Sh75}, completed by Gelbart--Jacquet \cite{GJ78}) gives the
        completed $L$-function $\Lambda(s, \Pi) = q(\Pi)^{s/2} L_\infty(s, \Pi) L(s, \Pi)$
        satisfying $\Lambda(s) = \varepsilon(\Pi) \Lambda(1-s, \tilde\Pi)$.
        From this, the density of zeros is controlled by $\log q(\Pi)$.
  \item \textbf{Siegel zero constant.}  From the Taylor expansion at a potential Siegel
        zero, the Rankin--Selberg bound (Theorem~F-2) gives
        $c_2 \geq c_4 \cdot \Res_{s=1}\zeta_Q(s) \cdot \min_v L_v(1, \Pi_v)^{-1}$,
        where the product is over finitely many bad places and is computable.
  \item \textbf{Interval arithmetic certification.}  Once $c_1$ and $c_2$ are expressed
        as explicit rational functions of the above quantities, \texttt{python-flint}/Arb
        will certify the numerical values with outward-rounded interval arithmetic.
\end{enumerate}

\section{Numerical Grounding: $N=1$ (Ramanujan $\Delta$)}

For $f = \Delta \in S_{12}(SL_2(\mathbb{Z}))$ (level $N=1$), the main theorem
is already verified by a direct computation ([THM F-3]):
\[
  L(1, \mathrm{sym}^2 \Delta) \approx 2.40578 \geq 2.405.
\]
Since $N = 1$, the bound $c_{\mathrm{eff}}/\log 1 = +\infty$ is vacuously satisfied.
The certification in \texttt{src/numerical\_delta.py} serves as a proof-of-concept
for the interval arithmetic infrastructure.
