\begin{abstract}
We outline the mollifier construction needed to establish the effective lower bound
$L(1, \mathrm{sym}^2 f) \geq c_{\mathrm{eff}}/\log N$.  The approach follows the strategy
of Goldfeld--Hoffstein--Lieman (1994), adapted to make the implicit constants explicit.
The key inputs are: (a) a mean value theorem for $GL_3$ $L$-functions, (b) Hecke
multiplicativity for symmetric square coefficients, and (c) the large sieve on $GL_3$.
All three are currently [OBL].
\end{abstract}

\section{The Symmetric Square $L$-Function as a $GL_3$ Automorphic Form}

By the theorem of Gelbart--Jacquet \cite{GJ78}, the symmetric square lift
$\Pi = \mathrm{sym}^2 \pi$ is an automorphic representation of $GL_3(\A_Q)$.
The $L$-function $L(s, \Pi)$ equals $L(s, \mathrm{sym}^2 f)$ defined by the Euler product.

\begin{definition}[Symmetric square Fourier coefficients]
The $n$-th Dirichlet coefficient of $L(s, \mathrm{sym}^2 f)$ is
\[
  a_{\Pi}(n) = \sum_{d^2 \mid n} \mu\Bigl(\frac{n}{d^2}\Bigr)
               a_f\Bigl(\frac{n}{d^2}\Bigr)\, a_f(n/d)\cdot (\text{local correction at bad primes})
\]
for squarefree $n$, with the full formula given by the Hecke multiplicativity relations
for $GL_3$.
\end{definition}

\begin{remark}
The key Hecke relation at unramified primes $p \nmid N$ is:
\[
  a_\Pi(p^k) = \tilde\alpha_p^{2k} + \tilde\alpha_p^{2k-2} + \cdots + \tilde\beta_p^{2k}
  = \chi_{2k}(\tilde\alpha_p, \tilde\beta_p) + \chi_{2k-2} + \cdots + \chi_0,
\]
where $\tilde\alpha_p \tilde\beta_p = 1$.  In particular $a_\Pi(p) = a_f(p)^2/p^{k-1} - 1$.
\end{remark}

\section{Mollifier Definition}

\begin{definition}[Mollifier]\label{def:mollifier}
Fix a mollifier length $X = N^\theta$ for some $\theta \in (0, 1/2)$ to be optimized.
Define the Dirichlet polynomial
\[
  M(s) = \sum_{n \leq X} \mu(n)\, a_\Pi(n)\, n^{-s}.
\]
Here $\mu$ is the Mobius function and $a_\Pi(n)$ are the Hecke eigenvalues of
$\Pi = \mathrm{sym}^2 f$.
\end{definition}

\begin{remark}
The mollifier $M(s)$ is designed so that $M(s) L(s, \Pi)$ is approximately 1
for $s$ near $1/2 + iT$: the partial sums truncate the Euler product, removing
the oscillation from $L$.
\end{remark}

\section{Mean Value Estimate [OBL]}

\begin{proposition}[Mean value, [OBL]]\label{prop:mean-value}
There exist absolute constants $c_0, T_0 > 0$ such that for all $T \geq T_0$:
\[
  \int_T^{2T} |M(\tfrac12+it)\, L(\tfrac12+it, \mathrm{sym}^2 f)|^2\, dt \geq c_0 T.
\]
\end{proposition}

\begin{remark}[What is needed to close this obligation]
This requires:
\begin{enumerate}
  \item A $GL_3$ Rankin--Selberg mean value: $\int_T^{2T} |L(1/2+it, \Pi)|^2 dt \sim cT$
        for $T \to \infty$ with explicit constant $c$.
  \item Hecke multiplicativity: the cross-term $\sum_{m,n \leq X} \mu(m)\mu(n)
        a_\Pi(m) \overline{a_\Pi(n)} (mn)^{-1/2}$ can be evaluated using
        $GL_3$ large sieve inequalities.
  \item Explicit constant tracking throughout.
\end{enumerate}
The key reference is Luo--Rudnick--Sarnak \cite{LRS95} for $GL_3$ mean values,
and Bruggeman--Motohashi or Blomer--Khan for explicit forms.
\end{remark}

\section{Zero-Density Estimate [OBL]}

\begin{proposition}[Quantitative exceptional-zero alternative, [OBL]]\label{prop:zero-density}
There exists an explicit $c_1 > 0$ such that $L(s, \mathrm{sym}^2 f)$ has at most
one real zero $\beta$ with $\beta > 1 - c_1/\log(N q_\infty)$, where $q_\infty$
is the Archimedean conductor.  Moreover, in the exceptional case one must prove
an explicit lower bound
\[
  L(1,\mathrm{sym}^2 f)\ge \frac{c_2}{\log(N q_\infty)}.
\]
\end{proposition}

\begin{remark}
The last estimate is not a consequence of pointwise residue positivity F-2.
It requires a uniform bound for the logarithmic derivative or for the Taylor
coefficient at the exceptional zero, with explicit dependence on the conductor.
That quantitative input is precisely still open.
\end{remark}

\begin{remark}[What is needed to close this obligation]
This requires:
\begin{enumerate}
  \item The Hadamard product representation for $L(s, \mathrm{sym}^2 f)$,
        giving $\log L'/L(s,\cdot) = \sum_\rho (s-\rho)^{-1} + \text{entire}$.
  \item An explicit bound on the analytic conductor $q(\Pi, s)$ for $\Pi = \mathrm{sym}^2 f$,
        needed to control the sum over zeros.
  \item Explicit constants (not just order-of-magnitude) in the standard
        log-free zero-free region.
\end{enumerate}
\end{remark}

\section{Connection to Lower Bound at $s=1$ [OBL]}

\begin{proposition}[Lower bound from mollifier, [OBL]]\label{prop:lbound-from-mollifier}
Assume Propositions \ref{prop:mean-value} and \ref{prop:zero-density}.  Then
for any primitive Hecke eigenform $f$ of prime level $p$ with no Siegel zero:
\[
  L(1, \mathrm{sym}^2 f) \geq \frac{c_2}{\log p}
\]
for an explicit constant $c_2 > 0$.
\end{proposition}

\begin{remark}[Proof sketch]
Using the zero-free region of width $\delta = c_1/\log p$ around $s=1$, standard
contour integration of $\log L(s, \mathrm{sym}^2 f)$ along the rectangle with
$\mathrm{Re}(s) \in [1-\delta, 1+\delta]$ and $|\mathrm{Im}(s)| \leq 2$ gives
\[
  \log L(1, \cdot) \geq -\frac{1}{\delta}\cdot C \log\log p + O(1)
                  \geq -C' \log\log p.
\]
This gives $L(1, \cdot) \geq p^{-\varepsilon}$ for any $\varepsilon > 0$, but the
logarithmic bound $c_2/\log p$ requires a sharper contour argument tracking
explicit constants.  The core research contribution is making $c_2$ explicit.
\end{remark}
